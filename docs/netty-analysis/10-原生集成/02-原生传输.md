# 原生传输层 C 代码

## 概述

Netty 的原生传输层 C 代码是 Java JNI 调用的底层实现，直接与 Linux/macOS 内核系统调用交互。代码分为两层：`transport-native-unix-common`（Unix 平台通用）和 `transport-native-epoll`（Linux epoll 专属）。这些 C 代码通过精心设计的 JNI 函数签名映射，将 Java 层的 `native` 方法桥接到内核系统调用。

**核心模块**: `transport-native-unix-common/src/main/c/`、`transport-native-epoll/src/main/c/`

**关键特性**:
- 完整的 Socket 生命周期管理（创建/绑定/监听/连接/关闭）
- epoll 事件驱动的 I/O 多路复用
- sendmmsg/recvmmsg 批量系统调用支持
- 零拷贝 splice/sendfile 支持
- eventfd/timerfd 高效事件通知

**非阻塞 Socket 创建**:

```c
int netty_unix_socket_nonBlockingSocket(int domain, int type, int protocol) {
#ifdef SOCK_NONBLOCK
    // Linux 2.6.27+ 支持 SOCK_NONBLOCK 标志，一次系统调用完成
    return socket(domain, type | SOCK_NONBLOCK, protocol);
#else
    // 降级方案：先创建再设置 O_NONBLOCK
    int socketFd = socket(domain, type, protocol);
    int flags = fcntl(socketFd, F_GETFL, 0);
    fcntl(socketFd, F_SETFL, flags | O_NONBLOCK);
    return socketFd;
#endif
}
```

**accept4 优化**:

```c
static jint netty_unix_socket_accept(JNIEnv* env, jclass clazz, jint fd, jbyteArray acceptedAddress) {
    for (;;) {
#ifdef SOCK_NONBLOCK
        if (accept4) {
            // accept4 一次调用设置 SOCK_NONBLOCK | SOCK_CLOEXEC
            socketFd = accept4(fd, &addr, &address_len, SOCK_NONBLOCK | SOCK_CLOEXEC);
        } else {
#endif
            socketFd = accept(fd, &addr, &address_len);
#ifdef SOCK_NONBLOCK
        }
#endif
        if (socketFd != -1) break;
        if (errno != EINTR) return -errno;
    }

    if (!accept4) {
        // accept4 不可用时需要额外两次 fcntl 调用
        fcntl(socketFd, F_SETFD, FD_CLOEXEC);
        fcntl(socketFd, F_SETFL, O_NONBLOCK);
    }
    return socketFd;
}
```

```c
// 自动检测 IPv4-mapped IPv6 地址
int netty_unix_socket_ipAddressLength(const struct sockaddr_storage* addr) {
    if (addr->ss_family == AF_INET) {
        return 4;  // 纯 IPv4
```

### 2. netty_unix_filedescriptor.c — 文件描述符操作

实现底层的 read/write/writev 操作，是所有 I/O 操作的基础。

**文件位置**: `transport-native-unix-common/src/main/c/netty_unix_filedescriptor.c`

**Java 类映射**: `io.netty.channel.unix.FileDescriptor`

**EINTR 自动重试**:

```c
static jint _write(JNIEnv* env, jclass clazz, jint fd, void* buffer, jint pos, jint limit) {
    ssize_t res;
    int err;
    do {
       res = write(fd, buffer + pos, (size_t)(limit - pos));
    } while (res == -1 && ((err = errno) == EINTR));  // 被信号中断则重试

    if (res < 0) {
        return -err;  // 返回负数 errno
    }
    return (jint) res;
}
```

**pipe2 兼容性处理**:

```c
// pipe2 是可选的，通过 weak symbol 检测
extern int pipe2(int pipefd[2], int flags)
    __attribute__((weak)) __attribute__((weak_import));

static jlong netty_unix_filedescriptor_newPipe(JNIEnv* env, jclass clazz) {
    int fd[2];
    if (pipe2) {
        pipe2(fd, O_NONBLOCK);  // 一次调用完成
    } else {
        pipe(fd);
        fcntl(fd[0], F_SETFL, flags | O_NONBLOCK);  // 降级方案
        fcntl(fd[1], F_SETFL, flags | O_NONBLOCK);
    }
    // 将两个 fd 编码到一个 64 位值中
    return (((jlong) fd[0]) << 32) | fd[1];
}
```

**JNI 版本优化**（直接访问字段 vs 方法调用）:

```c
jint netty_unix_filedescriptor_JNI_OnLoad(JNIEnv* env, const char* packagePrefix) {

}
```

### 3. netty_epoll_native.c — epoll 核心实现

Linux epoll 事件驱动的 I/O 多路复用实现。

**文件位置**: `transport-native-epoll/src/main/c/netty_epoll_native.c`

**Java 类映射**: `io.netty.channel.epoll.Native`

**epoll 创建与管理**:

```c
static jint netty_epoll_native_epollCreate(JNIEnv* env, jclass clazz) {
    jint efd;
    if (epoll_create1) {
        efd = epoll_create1(EPOLL_CLOEXEC);  // 原子设置 CLOEXEC
    } else {
        efd = epoll_create(126);
        fcntl(efd, F_SETFD, FD_CLOEXEC);     // 降级方案
    }
    return efd;
}
```

**epoll_wait 超时精确处理**:

```c
static inline jint netty_epoll_wait(JNIEnv* env, jint efd,
        struct epoll_event *ev, jint len, jint timeout) {
    int rc;
    if (timeout <= 0) {
        // 非阻塞或无限等待，直接调用
        while ((rc = epoll_wait(efd, ev, len, timeout)) < 0) {
            if (errno != EINTR) return -errno;
        }
    } else {
        // 有限超时：处理 EINTR 后的剩余时间
        struct timespec ts;
        clock_gettime(CLOCK_MONOTONIC, &ts);
        long deadline = ts.tv_sec * 1000 + ts.tv_nsec / 1000000 + timeout;

        while ((rc = epoll_wait(efd, ev, len, timeout)) < 0) {
            if (errno != EINTR) return -errno;
            clock_gettime(CLOCK_MONOTONIC, &ts);
            long now = ts.tv_sec * 1000 + ts.tv_nsec / 1000000;
            if (now >= deadline) return 0;
            timeout = deadline - now;  // 更新剩余超时
        }
    }
    return rc;
}
```

**epoll_pwait2 支持**（纳秒精度超时）:

```c
static inline jint netty_epoll_pwait2(JNIEnv *env, jint efd,
        struct epoll_event *ev, jint len, const struct timespec *timeout) {
    // epoll_pwait2 支持 timespec 精度，避免 timerfd_settime 的额外系统调用
    while ((rc = epoll_pwait2(efd, ev, len, timeout, NULL)) < 0) {
        if (errno != EINTR) return -errno;
    }
    return rc;
}
```

**eventfd 线程唤醒**:

```c
static jint netty_epoll_native_eventFd(JNIEnv* env, jclass clazz) {
    return eventfd(0, EFD_CLOEXEC | EFD_NONBLOCK);
}

static void netty_epoll_native_eventFdWrite(JNIEnv* env, jclass clazz, jint fd, jlong value) {
    for (;;) {
        jint ret = eventfd_write(fd, (eventfd_t) value);
        if (ret < 0 && errno == EAGAIN) {
            // 缓冲区满，先读取再写入
            eventfd_read(fd, &val);
            continue;
        }
        break;
    }
}
```

## 设计思想

### 1. Weak Symbol 兼容性检测

Netty 使用 GCC 的 `__attribute__((weak))` 特性检测可选系统调用：

```c
// 声明为 weak symbol，链接时如果不存在则为 NULL
extern int epoll_create1(int flags) __attribute__((weak));
extern int accept4(...) __attribute__((weak)) __attribute__((weak_import));

// 运行时检测
if (epoll_create1) {
    efd = epoll_create1(EPOLL_CLOEXEC);
} else {
    efd = epoll_create(126);
    fcntl(efd, F_SETFD, FD_CLOEXEC);
}
```

### 2. 错误处理模式

所有 JNI 函数遵循统一的错误处理模式：

```c
// 模式 1：返回负数 errno
static jint some_operation(...) {
    int res = syscall(...);
    if (res < 0) {
        return -errno;  // Java 层通过 Errors.ioResult() 处理
    }
    return res;
}

// 模式 2：抛出 Java 异常
static void some_setting(...) {
    if (setsockopt(...) < 0) {
        netty_unix_errors_throwChannelExceptionErrorNo(env, "setsockopt() failed: ", errno);
    }
}
```

### 3. EINTR 自动重试

所有阻塞系统调用都包含 EINTR 重试逻辑：

```c
do {
    res = some_blocking_syscall(...);
} while (res == -1 && ((err = errno) == EINTR));
```

### 4. 直接内存地址传递

使用 `intptr_t` 在 `jlong` 和原生指针之间安全转换：

```c
// Java long -> C 指针
struct epoll_event *ev = (struct epoll_event *)(intptr_t) address;

// 内存地址直接传递，避免 JNI 数组拷贝
void* buffer = (void *)(intptr_t) memoryAddress;
```

### epoll 事件处理完整流程


## 学习要点

1. **Weak Symbol 兼容性**：通过 `__attribute__((weak))` 检测可选系统调用（如 `accept4`、`epoll_create1`），在编译时不依赖特定内核版本，运行时自动降级

2. **直接系统调用**：对于 `sendmmsg`/`recvmmsg` 等较新的系统调用，使用 `syscall()` 直接调用而非依赖 GLIBC 封装，避免了对 GLIBC 版本的要求

3. **EINTR 处理**：所有阻塞系统调用都必须处理 `EINTR`（被信号中断），Netty 使用 `do-while` 循环自动重试

4. **超时精确性**：epoll_wait 的超时在 EINTR 后需要重新计算剩余时间，使用 `clock_gettime(CLOCK_MONOTONIC)` 获取单调时钟避免时钟跳变

5. **epoll_pwait2 优化**：对于纳秒精度的超时，优先使用 `epoll_pwait2` 避免额外的 `timerfd_settime` 系统调用，减少系统调用开销
